Kennzeichnet ein Zeichenfolgenfeld als Sprachcode. Der Inspector zeigt daraufhin ein Dropdown-Menü mit den Sprachen an, die tatsächlich in den VoiceLineData-Assets des Projekts vorhanden sind (zuzüglich der Ausgangssprache), anstatt eines Freitextfeldes, in das ein Code manuell eingegeben werden muss.

public sealed class LanguageCodeAttribute : PropertyAttribute

object
LanguageCodeAttribute

Examples

[SerializeField, LanguageCode] private string _menuLanguage = "fr";

public void ApplyLanguage()
{
    SubVox.CurrentLanguage = _menuLanguage;
}
