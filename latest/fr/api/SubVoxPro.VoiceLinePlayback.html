Demande de lecture d’une voiceline, renvoyée par Play(VoiceLineData, EVoiceLinePriority, EVoiceLinePlayMode) et Play(VoiceLineData, EVoiceLinePriority, EVoiceLinePlayMode) : consultez son State, arrêtez-la ou attendez sa fin dans une coroutine avec yield return.

public sealed class VoiceLinePlayback : CustomYieldInstruction

object

Examples

private IEnumerator Dialogue()
{
    yield return SubVox.Play(question);
    yield return new WaitForSeconds(0.5f);

    VoiceLinePlayback answer = SubVox.Play(reply);
    answer.Finished += done => Debug.Log($"Answer ended: {done.State}");
    yield return answer;
}
