Eine Anfrage zum Abspielen einer Voice Line, die von Play(VoiceLineData, EVoiceLinePriority, EVoiceLinePlayMode) und Play(VoiceLineData, EVoiceLinePriority, EVoiceLinePlayMode) zurückgegeben wird: Verfolgen Sie deren Status unter State, stoppen Sie sie oder warten Sie in einer Coroutine mit yield return darauf.

public sealed class VoiceLinePlayback : CustomYieldInstruction
Vererbung
object
VoiceLinePlayback

Examples

private IEnumerator Dialogue()
{
    yield return SubVox.Play(question);
    yield return new WaitForSeconds(0.5f);

    VoiceLinePlayback answer = SubVox.Play(reply);
    answer.Finished += done => Debug.Log($"Answer ended: {done.State}");
    yield return answer;
}
