Interface IVoiceLineAudioClock

Namespace
SubVoxPro
Assembly
SubVoxPro.dll

Audio, das Sie außerhalb der „AudioSource“ von Unity abspielen (FMOD, Wwise, ein Video …), gefolgt von PlayExternal(VoiceLineData, IVoiceLineAudioClock, EVoiceLinePriority, EVoiceLinePlayMode), damit die Untertitel und Events damit synchron bleiben.

public interface IVoiceLineAudioClock

Examples

public sealed class FmodVoiceClock : IVoiceLineAudioClock
{
    public FMOD.Studio.EventInstance Instance;

    public float Time
    {
        get { Instance.getTimelinePosition(out int milliseconds); return milliseconds / 1000f; }
    }

    public bool IsPlaying
    {
        get { Instance.getPlaybackState(out FMOD.Studio.PLAYBACK_STATE state); return state != FMOD.Studio.PLAYBACK_STATE.STOPPED; }
    }
}

emitter.PlayExternal(voiceLine, new FmodVoiceClock { Instance = instance });
instance.start();
