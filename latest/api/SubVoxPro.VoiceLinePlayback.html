One request to play a voice line, returned by Play(VoiceLineData, EVoiceLinePriority, EVoiceLinePlayMode) and Play(VoiceLineData, EVoiceLinePriority, EVoiceLinePlayMode): follow its State, stop it, or wait for it in a coroutine with yield return.

public sealed class VoiceLinePlayback : CustomYieldInstruction

Examples

private IEnumerator Dialogue()
{
    yield return SubVox.Play(question);
    yield return new WaitForSeconds(0.5f);

    VoiceLinePlayback answer = SubVox.Play(reply);
    answer.Finished += done => Debug.Log($"Answer ended: {done.State}");
    yield return answer;
}
